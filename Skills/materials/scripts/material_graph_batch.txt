# material_graph_batch.txt — Build a graph node by node, driven from a table.
#
# Sample script for the materials skill. Run via execute_python.
#
# UE 5.9: there is NO batch API any more. batch_create_expressions, batch_create_specialized,
# batch_connect_expressions and batch_set_properties were all removed with the rest of
# MaterialNodeService's graph API. Epic's AddExpression / ConnectExpressions / SetProperties are
# one node and one wire per call — so drive them from a table and the graph is still one block.
#
# Epic's AddExpression + SetProperties({"MaterialFunction": ...}) produces a node with NO PINS:
# the generic property writer skips PostEditChangeProperty. Native set_material_function populates
# them. MaterialNodeService keeps create_collection_parameter (not needed in this script).
import json, unreal, vibeue

MT = "EditorToolset.MaterialToolset"
OT = "EditorToolset.ObjectToolset"
AT = "EditorToolset.AssetToolset"
E  = "/Script/Engine."

PATH = "/Game/Materials/M_Character"   # must already exist


def ref(p):
    tail = p.rsplit("/", 1)[-1]
    return {"refPath": p if "." in tail else "{}.{}".format(p, tail)}


def add(cls, x, y):
    return vibeue.exec_tool(MT, "AddExpression", {
        "materialOrFunction": ref(PATH), "expressionClass": {"refPath": E + cls},
        "x": x, "y": y})


def set_props(obj, values):
    return vibeue.exec_tool(OT, "SetProperties", {"instance": obj, "values": json.dumps(values)})


if not vibeue.exec_tool(AT, "Exists", {"path": PATH}):
    raise RuntimeError("material missing: " + PATH)

# TintColor (Vector) * Intensity (Scalar) -> BaseColor, plus a function call on Normal.
tint = add("MaterialExpressionVectorParameter", -600, 0)
set_props(tint, {"ParameterName": "TintColor", "Group": "Surface",
                 "DefaultValue": {"r": 1.0, "g": 1.0, "b": 1.0, "a": 1.0}})

inten = add("MaterialExpressionScalarParameter", -600, 150)
set_props(inten, {"ParameterName": "Intensity", "Group": "Surface", "DefaultValue": 1.0})

mult = add("MaterialExpressionMultiply", -300, 50)

# GetExpressionInputNames returns exactly the names ConnectExpressions accepts — no display
# labels, no type suffixes. A connect that does not land RAISES instead of wiring nothing.
print("Multiply inputs:", vibeue.exec_tool(MT, "GetExpressionInputNames", {"expression": mult}))

for src, pin in ((tint, "A"), (inten, "B")):
    vibeue.exec_tool(MT, "ConnectExpressions", {
        "fromExpression": src, "fromOutputName": "", "toExpression": mult, "toInputName": pin})

vibeue.exec_tool(MT, "ConnectToOutput", {"expression": mult, "outputName": "",
                                         "materialProperty": "MP_BaseColor"})

# Native Python creates and initializes a function call node with real pins.
fn = unreal.MaterialEditingLibrary.create_material_expression(
    unreal.load_asset(PATH), unreal.MaterialExpressionMaterialFunctionCall, -600, 350)
fn.set_material_function(unreal.load_asset(
    "/Engine/Functions/Engine_MaterialFunctions02/Utility/BlendAngleCorrectedNormals"))
fn_ref = {"refPath": fn.get_path_name()}
print("function call pins:", vibeue.exec_tool(MT, "GetExpressionInputNames", {"expression": fn_ref}),
      "->", vibeue.exec_tool(MT, "GetExpressionOutputNames", {"expression": fn_ref}))
vibeue.exec_tool(MT, "ConnectToOutput", {"expression": fn_ref, "outputName": "",
                                         "materialProperty": "MP_Normal"})

vibeue.exec_tool(MT, "LayoutExpressions", {"materialOrFunction": ref(PATH)})
vibeue.exec_tool(MT, "Recompile", {"materialOrFunction": ref(PATH)})   # raises on a compile error
vibeue.exec_tool(AT, "SaveAsset", {"assetPath": PATH})

stats = vibeue.exec_tool(MT, "GetStatistics", {"material": ref(PATH)})
print("samplers:", stats["num_samplers"],
      "pixel texture samples:", stats["num_pixel_texture_samples"],
      "pixel instructions:", stats["num_pixel_shader_instructions"])
