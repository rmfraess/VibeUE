# scatter_foliage.txt — Scatter foliage instances at points.
#
# Sample script for the foliage skill. Run via execute_python.
import unreal
fs = unreal.FoliageService

MESH = "/Game/Meshes/SM_Bush"
# For a persistent foliage type, use native AssetTools.create_asset with
# FoliageType_InstancedStaticMeshFactory and set the mesh on the created asset.

# Scatter at explicit world locations (use trace_to_surface to drop onto the landscape)
locations = [unreal.Vector(x*200.0, 0.0, 0.0) for x in range(5)]
r = fs.add_foliage_instances(MESH, locations, min_scale=0.9, max_scale=1.1,
                             align_to_normal=True, random_yaw=True, trace_to_surface=True)
world = unreal.get_editor_subsystem(unreal.UnrealEditorSubsystem).get_editor_world()
mesh = unreal.load_asset(MESH)
count = sum(c.get_instance_count() for a in unreal.GameplayStatics.get_all_actors_of_class(world, unreal.InstancedFoliageActor)
            for c in a.get_components_by_class(unreal.InstancedStaticMeshComponent) if c.static_mesh == mesh)
print("scatter:", r, "count:", count)
