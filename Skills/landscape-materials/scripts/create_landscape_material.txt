# create_landscape_material.txt — Create a landscape material with a layer blend, compile, assign.
#
# Sample script for the landscape-materials skill. Run via execute_python.
#
# SPLIT: compiling is Epic's MaterialToolset.Recompile (VibeUE's MaterialService is GONE) and is
# slow, so it is a SEPARATE call_tool step — never in this block. Always create LayerInfo objects
# for layers.
import json
import unreal

# The material asset and every node in it are Epic's MaterialToolset / ObjectToolset
# (expression refs are {"refPath": ...} handles).
path = "/Game/Landscape/M_Landscape"
MAT = {"refPath": "%s.%s" % (path, path.rsplit("/", 1)[-1])}
if not await call_tool("EditorToolset.AssetToolset.Exists", assetPath=path):
    r = await call_tool("EditorToolset.MaterialToolset.CreateMaterial",
                        folderPath="/Game/Landscape", assetName="M_Landscape")
    print("CREATED:", r)

# Build a LayerBlend node with named layers (Grass, Rock); wire to BaseColor and textures afterwards (see workflows.md).
blend = await call_tool("EditorToolset.MaterialToolset.AddExpression", materialOrFunction=MAT,
                        expressionClass={"refPath": "/Script/Landscape.MaterialExpressionLandscapeLayerBlend"}, x=-400, y=0)
await call_tool("EditorToolset.ObjectToolset.SetProperties", instance=blend, values=json.dumps({"Layers": [
    {"LayerName": "Grass", "BlendType": "LB_WeightBlend", "PreviewWeight": 1.0},
    {"LayerName": "Rock", "BlendType": "LB_WeightBlend", "PreviewWeight": 1.0}]}))

unreal.EditorAssetLibrary.save_asset(path)
print("done — now run the two engine steps, each as its own call_tool:")
print('  call_tool MaterialToolset.ConnectToOutput '
      '{"expression": %s, '
      '"outputName": "", "materialProperty": "MP_BaseColor"}' % json.dumps(blend))
print('  call_tool MaterialToolset.Recompile {"materialOrFunction": {"refPath": "%s.%s"}}'
      % (path, path.rsplit("/", 1)[-1]))
