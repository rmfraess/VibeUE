# create_sound_cue.txt — Create a SoundCue and add a delay node.
#
# Sample script for the sound-cues skill. Run via execute_python.
import unreal
scs = unreal.get_editor_subsystem(unreal.SoundCueEditorSubsystem)
ASSET = "/Game/Audio/SC_SkillTest"
assert not unreal.EditorAssetLibrary.does_asset_exist(ASSET), ASSET
cue = scs.create_sound_cue("/Game/Audio", "SC_SkillTest")
assert cue
delay = scs.add_node(cue, unreal.SoundNodeDelay)
assert delay
scs.set_root_node(cue, delay)
unreal.EditorAssetLibrary.save_asset(ASSET)
